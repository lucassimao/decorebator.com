  // GetAllWordlistsProgress retrieves progress summary for all user's wordlists
  func (r *AnalyticsRepository) GetAllWordlistsProgress(ctx context.Context, userID int64) ([]model.WordlistProgress, error) {
      query := `
          WITH wordlist_stats AS (
              SELECT 
                  w.wordlist_id,
                  wl.name as wordlist_name,
                  wl.language_code,
                  COUNT(DISTINCT w.id) as total_words,
                  COUNT(DISTINCT CASE WHEN wm.mastery_level >= 0.8 THEN w.id END) as words_mastered,
                  CASE 
                      WHEN COUNT(DISTINCT w.id) > 0 
                      THEN ROUND(AVG(COALESCE(wm.mastery_level, 0)) * 100, 2)
                      ELSE 0 
                  END as progress_percent,
                  MAX(lp.date) as last_activity_date
              FROM words w
              JOIN wordlists wl ON w.wordlist_id = wl.id
              LEFT JOIN word_mastery wm ON w.id = wm.word_id AND w.user_id = wm.user_id
              LEFT JOIN learning_progress lp ON w.wordlist_id = lp.wordlist_id AND w.user_id = lp.user_id
              WHERE w.user_id = $1 
                  AND w.learned = FALSE
                  AND wl.deleted_at IS NULL
              GROUP BY w.wordlist_id, wl.name, wl.language_code
          ),
          streaks AS (
              SELECT 
                  wordlist_id,
                  COUNT(*) as current_streak
              FROM (
                  SELECT 
                      wordlist_id,
                      date,
                      date - (ROW_NUMBER() OVER (PARTITION BY wordlist_id ORDER BY date DESC))::int AS streak_group
                  FROM learning_progress
                  WHERE user_id = $1 
                      AND total_quiz_attempts > 0
                      AND date >= CURRENT_DATE - INTERVAL '30 days'
              ) streak_calc
              WHERE streak_group = (
                  SELECT streak_group 
                  FROM (
                      SELECT 
                          date - (ROW_NUMBER() OVER (ORDER BY date DESC))::int AS streak_group
                      FROM learning_progress
                      WHERE user_id = $1 
                          AND wordlist_id = streak_calc.wordlist_id
                          AND total_quiz_attempts > 0
                  ) sg
                  WHERE date = CURRENT_DATE
                  LIMIT 1
              )
              GROUP BY wordlist_id
          )
          SELECT 
              ws.wordlist_id,
              ws.wordlist_name,
              ws.language_code,
              ws.total_words,
              ws.words_mastered,
              ws.progress_percent,
              COALESCE(s.current_streak, 0) as current_streak,
              ws.last_activity_date
          FROM wordlist_stats ws
          LEFT JOIN streaks s ON ws.wordlist_id = s.wordlist_id
          ORDER BY ws.wordlist_id;
      `

      rows, err := r.db.Query(ctx, query, userID)
      if err != nil {
          return nil, fmt.Errorf("failed to get wordlists progress: %w", err)
      }
      defer rows.Close()

      var progress []model.WordlistProgress
      for rows.Next() {
          var wp model.WordlistProgress
          err := rows.Scan(
              &wp.WordlistID,
              &wp.WordlistName,
              &wp.LanguageCode,
              &wp.TotalWords,
              &wp.WordsMastered,
              &wp.ProgressPercent,
              &wp.CurrentStreak,
              &wp.LastActivityDate,
          )
          if err != nil {
              return nil, err
          }
          progress = append(progress, wp)
      }

      return progress, rows.Err()
  }

  Created: api/internal/common/redis.go

  package common

  import (
      "context"
      "fmt"
      "os"
      "strconv"
      "sync"
      "time"

      "github.com/redis/go-redis/v9"
  )

  var (
      redisClient *redis.Client
      redisOnce   sync.Once
  )

  // GetRedisClient returns a singleton Redis client instance
  func GetRedisClient() (*redis.Client, error) {
      var initErr error

      redisOnce.Do(func() {
          host := os.Getenv("REDIS_HOST")
          if host == "" {
              host = "localhost"
          }

          port := os.Getenv("REDIS_PORT")
          if port == "" {
              port = "6379"
          }

          password := os.Getenv("REDIS_PASSWORD")

          db := 0
          if dbStr := os.Getenv("REDIS_DB"); dbStr != "" {
              if parsed, err := strconv.Atoi(dbStr); err == nil {
                  db = parsed
              }
          }

          poolSize := 100
          if poolStr := os.Getenv("REDIS_POOL_SIZE"); poolStr != "" {
              if parsed, err := strconv.Atoi(poolStr); err == nil {
                  poolSize = parsed
              }
          }

          redisClient = redis.NewClient(&redis.Options{
              Addr:         fmt.Sprintf("%s:%s", host, port),
              Password:     password,
              DB:           db,
              PoolSize:     poolSize,
              MinIdleConns: 10,
              MaxRetries:   3,
          })

          // Test connection
          ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
          defer cancel()

          if err := redisClient.Ping(ctx).Err(); err != nil {
              initErr = fmt.Errorf("failed to connect to Redis: %w", err)
              redisClient = nil
          }
      })

      if initErr != nil {
          return nil, initErr
      }

      if redisClient == nil {
          return nil, fmt.Errorf("Redis client not initialized")
      }

      return redisClient, nil
  }

  ### 3.1 Add Cache Invalidation to SaveQuizResult ✅ COMPLETED

  **Goal**: Invalidate analytics caches for premium users after quiz completion for real-time updates

  **What was implemented**:

  Updated `api/internal/service/leitner_system_strategy.go` to add cache invalidation after quiz completion:

  ```go
  // In SaveQuizResult function, after analytics tracking succeeds:
  
  // Invalidate analytics cache for premium users
  go func() {
      ctx := context.Background()
      db, err := common.GetDBConnection()
      if err != nil {
          common.Logger.Error("Failed to get DB connection for cache invalidation", "error", err)
          return
      }
      
      // Get user to check subscription
      userRepo := &repository.UserRepository{Db: db}
      user, err := userRepo.Find(ctx, userID)
      if err != nil {
          common.Logger.Error("Failed to get user for cache invalidation", "error", err)
          return
      }
      
      // Only invalidate for non-free users
      if user.SubscriptionPlan != model.SubscriptionPlanFree {
          cache, err := common.GetRedisClient()
          if err != nil {
              common.Logger.Warn("Redis not available for cache invalidation", "error", err)
              return
          }
          
          cachedService := &service.CachedAnalyticsService{
              // Exported fields for cache invalidation
          }
          cachedService.SetCache(cache)
          
          // Invalidate wordlist-specific caches
          if err := cachedService.InvalidateWordlistAnalytics(ctx, userID, wordlistID); err != nil {
              common.Logger.Error("Failed to invalidate analytics cache", "error", err)
          } else {
              common.Logger.Info("Analytics cache invalidated for premium user",
                  "userID", userID,
                  "wordlistID", wordlistID,
                  "plan", user.SubscriptionPlan)
          }
      }
  }()
  ```

  **Key implementation details**:
  - Runs asynchronously in goroutine to avoid blocking quiz response
  - Only invalidates cache for premium users (non-free plans)
  - Gracefully handles Redis unavailability
  - Logs operations for monitoring
  - Retrieves wordlistID from the word data within the same transaction

  ## 🔧 Troubleshooting

  ### Common Issues:

  #### 1. Redis Connection Failures
  - **Symptom**: Logs show "Redis not available, analytics will use database only"
  - **Impact**: System falls back to database queries (still works, but slower)
  - **Fix**: 
    - Check Redis connection string in .env
    - Verify Redis is running: `redis-cli ping`
    - Check network connectivity

  #### 2. High Database Load After Deployment
  - **Symptom**: Database CPU spikes, slow queries
  - **Possible Causes**:
    - Missing indexes (run migration 000044)
    - Cache not working (check Redis)
    - Too many service instances being created
  - **Fix**:
    - Verify indexes: `\d+ word_mastery` in psql
    - Monitor service instantiation in logs

  #### 3. Stale Data for Premium Users
  - **Symptom**: Premium users see old data despite 1-minute cache
  - **Check**:
    - User subscription status detection
    - Cache invalidation after quiz completion
    - TTL being applied correctly
  - **Debug**: Add logging to track cache TTL application

  ### Debug Logging:

  ```go
  // Add to CachedAnalyticsService methods
  ```

  ## Implementation Notes

  ### What We Changed:

  1. **Database Performance**:
     - Added 6 strategic indexes for analytics queries
     - Created batch endpoint to fetch all wordlists progress in one query

  2. **Caching Architecture**:
     - Redis caching with tier-based TTLs (1hr free, 1min premium)

     - Graceful fallback when Redis unavailable

  3. **API Design**:
     - New `/analytics/progress-summary` endpoint
     - Factory pattern: `NewAnalyticsServiceForUser(plan)`
     - Interface-based design for flexibility

  ### What We Didn't Change:

  1. **Kept Existing Endpoints**: All current analytics endpoints remain unchanged
  2. **Database Schema**: No table changes, only added indexes
  3. **Mobile Compatibility**: New batch endpoint is additive, old app versions continue working

  ### Key Architectural Decisions:

  1. **Per-Request Service Instantiation**: 
     - Each request creates a service instance with user's plan
     - Avoids global state and concurrency issues
     - Clean separation of concerns

  2. **Cache Key Design**:
     - Pattern: `analytics:type:userID:wordlistID`
     - Enables targeted invalidation
     - Clear namespace separation

  3. **Graceful Degradation**:
     - System works without Redis (just slower)
     - No hard dependencies on cache layer
     - Automatic fallback behavior
